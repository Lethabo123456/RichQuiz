• Log-Normal Distribution: A Log-Normal distribution is another type of continuous 
distribution of logarithmic values that form a normal distribution. We can transform a 
log-normal distribution into a normal distribution. 

Here is a code example showing the use of Log-Normal Distribution 
import matplotlib.pyplot as plt 
import numpy as np 
  
def lognormalDist(): 
    mu, sigma = 3, 1 
    s = np.random.lognormal(mu, sigma, 1000) 
     
    # Generate the histogram 
    cnt, bins, ignored = plt.hist(s, 80, density=True, align='mid', color='y') 
     
    # Generate points on the x-axis 
    x = np.linspace(min(bins), max(bins), 10000) 
     
    # Calculate the lognormal distribution 
    calc = (np.exp(-(np.log(x) - mu) ** 2 / (2 * sigma ** 2)) 
            / (x * sigma * np.sqrt(2 * np.pi))) 
    
    # Plot the lognormal distribution 
    plt.plot(x, calc, linewidth=2.5, color='g') 
     
    # Adjust the plot 
    plt.axis('tight') 
     
    # Show the plot 
    plt.show() 
  
lognormalDist()
