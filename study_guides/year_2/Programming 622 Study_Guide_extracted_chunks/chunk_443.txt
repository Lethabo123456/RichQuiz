bool queueType<Type>::isEmptyQueue() const  
{  
return (count == 0);  
} //end isEmptyQueue  
  
template <class Type>  
bool queueType<Type>::isFullQueue() const  
{  
return (count == maxQueueSize);  
} //end isFullQueue
