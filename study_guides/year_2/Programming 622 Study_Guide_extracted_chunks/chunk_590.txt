urrent->info < firstOutOfOrder->info)  
{ trailCurrent = current;

}

= current;  
trailCurrent->link = firstOutOfOrder;  
}  
else lastInOrder = lastInOrder->link;  
}  
} //end while  
} //end linkedInsertionSort
