Instead of waiting until all tasks are finished, you can also wait until one of the tasks is finished. You use 
the WaitAny method for this. example 5-14 shows how this works. 
 
using System.Threading;
using System.Threading.Tasks; 
using System; 
 
namespace Chapter1 
{ 
    public static class Program 
    { 
        public static void Main() 
        { 
            Task[] tasks = new Task[3]; 
            tasks[0] = Task.Run(() => 
            { 
                Thread.Sleep(1000); 
                Console.WriteLine("1"); 
                return 1; 
            }); 
            tasks[1] = Task.Run(() => 
            { 
                Thread.Sleep(1000); 
                Console.WriteLine("2"); 
                return 2; 
            }); 
            tasks[2] = Task.Run(() => 
            { 
                Thread.Sleep(1000); 
                Console.WriteLine("3"); 
                return 3; 
            } 
            ); 
            Task.WaitAll(tasks); 
        } 
    } 
}
