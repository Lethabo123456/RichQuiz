    <script> 
        // Raft visualization state 
        let nodes = []; 
        let currentTerm = 1; 
        let messages = []; 
         
        // Node states 
        const NodeState = { 
            FOLLOWER: 'follower', 
            CANDIDATE: 'candidate', 

            LEADER: 'leader' 
        }; 
         
        // Initialize nodes 
        function initializeNodes() { 
            const cluster = document.getElementById('cluster'); 
             
            for (let i = 1; i <= 5; i++) { 
                const nodeData = { 
                    id: i, 
                    state: NodeState.FOLLOWER, 
                    term: currentTerm, 
                    votes: 0 
                }; 
                nodes.push(nodeData); 
                 
                const nodeElement = document.createElement('div'); 
                nodeElement.className = `node ${NodeState.FOLLOWER}`; 
                nodeElement.id = `node-${i}`; 
                nodeElement.innerHTML = ` 
                    <div>Node ${i}</div> 
                    <div>Follower</div> 

                    <div>Term: ${currentTerm}</div> 
                `; 
                nodeElement.onclick = () => nodeClicked(i); 
                 
                cluster.appendChild(nodeElement); 
            } 
        } 
         
        // Update node display 
        function updateNode(nodeId) { 
            const node = nodes.find(n => n.id === nodeId); 
            const element = document.getElementById(`node-${nodeId}`); 
             
            element.className = `node ${node.state}`; 
            element.innerHTML = ` 
                <div>Node ${nodeId}</div> 
                <div>${node.state.charAt(0).toUpperCase() + node.state.slice(1)}</div> 
                <div>Term: ${node.term}</div> 
            `; 
        } 
         
        // Handle node clicks 

        function nodeClicked(nodeId) { 
            const node = nodes.find(n => n.id === nodeId); 
            logMessage(` Clicked Node ${nodeId} (${node.state})`); 
        } 
         
        // Start election process 
        function startElection() { 
            logMessage(" Starting election process..."); 
             
            // Reset all nodes to followers 
            nodes.forEach(node => { 
                node.state = NodeState.FOLLOWER; 
                node.votes = 0; 
                updateNode(node.id); 
            }); 
             
            // Random node becomes candidate 
            const candidateId = Math.floor(Math.random() * 5) + 1; 
            const candidate = nodes.find(n => n.id === candidateId); 
             
            candidate.state = NodeState.CANDIDATE; 
            candidate.term = ++currentTerm; 

            candidate.votes = 1; // Vote for self 
             
            updateNode(candidateId); 
            document.getElementById('current-term').textContent = currentTerm; 
             
            logMessage(` Node ${candidateId} became candidate for term ${currentTerm}`); 
             
            // Simulate voting process 
            setTimeout(() => simulateVoting(candidateId), 1000); 
        }
